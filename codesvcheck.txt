module top #(parameter N = 8)
(
input  logic [N-1:0] i_P_INPUT,
input  logic    i_V_INPUT,
input  logic    i_CLK,
input  logic   i_RST,
input  logic    i_P_EN,
input  logic  i_P_BIT,
output logic   o_Tx_output,
output logic   o_BUSY
);
logic SERIAL_OUT;
logic [1:0]  mux_sel;
logic  SHT_EN;
logic  PARITY_BIT;
logic  start;
logic  stop;
logic  accept;
assign start  = 1'b0;
assign stop   = 1'b1;
assign accept = i_V_INPUT & ~o_BUSY;
// i named them the same ^^
serializer #(.N(N)) serializer_top(
.i_P_INPUT(i_P_INPUT),
.i_V_INPUT(accept),
.i_SHT_EN(SHT_EN),
.i_CLK(i_CLK),
.i_RST(i_RST),
.o_SERIAL_OUT(SERIAL_OUT)
);
FSM FSM_top(
.i_V_INPUT(i_V_INPUT),
.i_P_EN(i_P_EN),
.i_CLK(i_CLK),
.i_RST(i_RST),
.o_BUSY(o_BUSY),
.o_SHT_EN(SHT_EN),
.o_mux_sel(mux_sel)
);
parity #(.N(N)) parity_top(
.i_P_INPUT(i_P_INPUT),
.i_P_EN(i_P_EN),
.i_P_BIT(i_P_BIT),
.o_PARITY_BIT(PARITY_BIT)
);
MUX4 MUX_top(
.i_start(start),
.i_stop(stop),
.i_SERIAL_OUT(SERIAL_OUT),
.i_PARITY_BIT(PARITY_BIT),
.i_mux_sel(mux_sel),
.o_Tx_output(o_Tx_output)
);
endmodule


module MUX4(
input logic i_start,
input logic i_stop,
input logic i_SERIAL_OUT,
input logic i_PARITY_BIT,
input logic [1:0] i_mux_sel,
output logic o_Tx_output
);
always_comb begin
case(i_mux_sel)
2'b00:
    o_Tx_output = i_start;
2'b01:
    o_Tx_output = i_SERIAL_OUT;
2'b10:
    o_Tx_output = i_PARITY_BIT;
2'b11:
    o_Tx_output = i_stop;
endcase
end
endmodule

module serializer #(
    parameter N = 8
)(
    input  logic [N-1:0] i_P_INPUT,
    input  logic   i_V_INPUT,
    input  logic   i_SHT_EN,
    input  logic    i_CLK,
    input  logic     i_RST,
    output logic    o_SERIAL_OUT
);
logic [N-1:0] shift_reg;
    always_ff @(posedge i_CLK or negedge i_RST)
    begin
        if(i_RST==0) begin
            shift_reg <= {N{1'b0}};
        end
        else if(i_V_INPUT==1) begin
            shift_reg <= i_P_INPUT;
        end
        else if(i_SHT_EN==1) begin
            shift_reg <= {1'b0, shift_reg[N-1:1]};
        end
    end
    assign o_SERIAL_OUT = shift_reg[0];
endmodule


module parity #(
    parameter N = 8
)(
input logic [N-1:0] i_P_INPUT,
input logic       i_P_EN,
input logic        i_P_BIT,
output logic    o_PARITY_BIT
);
int i;
int count_ones;
always_comb begin
count_ones = 0;
o_PARITY_BIT = 0;
        for (i = 0; i < N; i = i + 1) begin
            if (i_P_INPUT[i] == 1)
                count_ones = count_ones + 1;
        end
        if (i_P_EN == 1) begin
            if (i_P_BIT == 0) begin
                if (count_ones % 2 == 0)
                    o_PARITY_BIT = 0;
                else
                    o_PARITY_BIT = 1;
            end
            else begin
                if (count_ones % 2 == 0)
                    o_PARITY_BIT = 1;
                else
                    o_PARITY_BIT = 0;
            end
        end
    end
endmodule



module FSM(
input  logic    i_V_INPUT,
input  logic   i_P_EN,
input  logic    i_CLK,
input  logic    i_RST,
output logic    o_BUSY,
output logic    o_SHT_EN,
output logic [1:0] o_mux_sel
);
parameter logic [2:0] S0 = 3'b000;
parameter logic [2:0] S1 = 3'b001;
parameter logic [2:0] S2 = 3'b010;
parameter logic [2:0] S3 = 3'b011;
parameter logic [2:0] S4 = 3'b100;
logic [2:0] state;
logic [2:0] next_state;
logic [2:0] count;
always_ff @(posedge i_CLK or negedge i_RST)
begin
if (i_RST == 0)
begin
   state <= S0;
   count <= 0;
end
else
begin
   state <= next_state;
   if (state == S2)
      count <= count + 1;
   else
      count <= 0;
end
end
always_comb
begin
next_state = state;
o_mux_sel    = 2'b00;
o_SHT_EN     = 0;
o_BUSY       = (state == S0) ? 1'b0 : 1'b1;
case(state)
S0: begin 
    if (i_V_INPUT == 1) 
        next_state = S1;
    o_SHT_EN = 0;
    o_mux_sel = 2'b11;
end
// then when start bit finish the serializer start work
S1: begin 
    next_state = S2;
    o_SHT_EN = 0;
    o_mux_sel = 2'b00;
end
// data bits
S2:
begin
o_SHT_EN = 1;
o_mux_sel = 2'b01;
// after 8 data bits
if (count == 7)
begin
    if (i_P_EN == 1)
    begin
         next_state = S3;
    end
    else
    begin
         next_state = S4;
    end
end
end
// parity
S3: begin 
    next_state = S4;
    o_SHT_EN = 0;
    o_mux_sel = 2'b10;
end
// stop
S4: begin 
    next_state = S0;
    o_SHT_EN = 0;
    o_mux_sel = 2'b11;
end
endcase
end
endmodule
